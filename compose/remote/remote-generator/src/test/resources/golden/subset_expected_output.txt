/**
 * [RemoteOperationSink] implementation that writes wire bytes directly into [buffer] without
 * allocating intermediate [OperationNode] instances.
 *
 * @param buffer destination [WireBuffer]
 * @param targetApiLevel target wire protocol API level (defaults to 8)
 * @param supportedOpcodes optional set of allowed opcodes for profile validation
 * @param onOpcode optional callback invoked before each opcode is written
 */
@RestrictTo(RestrictTo.Scope.LIBRARY_GROUP)
public class WireBufferOperationSink(
    public val buffer: WireBuffer = WireBuffer(),
    public val targetApiLevel: Int = DEFAULT_DOCUMENT_API_LEVEL,
    public val supportedOpcodes: Set<Int>? = null,
    private val onOpcode: ((Int) -> Unit)? = null,
) : RemoteOperationSink {
    /** Total bytes written to [buffer] so far. */
    public val bytesWritten: Int
        get() = buffer.size()

    private fun validateOpcode(opcode: Int, operationName: String) {
        val allowed = supportedOpcodes
        if (allowed != null && opcode !in allowed) {
            throw IllegalArgumentException(
                "Operation $operationName (opcode=$opcode) is not supported by active profile"
            )
        }
        onOpcode?.invoke(opcode)
    }

    /**
     * Starts a `Skip` (`236`) block with a placeholder byte-length field and returns the buffer
     * offset to pass to [endSkip].
     */
    public fun beginSkip(conditionType: Int, `value`: Int): Int {
        validateOpcode(SkipNode.OPCODE, "Skip")
        buffer.start(SkipNode.OPCODE)
        buffer.writeInt(conditionType)
        buffer.writeInt(value)
        val skipOffset = buffer.index
        buffer.writeInt(0)
        return skipOffset
    }

    /** Backpatches the skipped byte length at [skipOffset] opened by [beginSkip]. */
    public fun endSkip(skipOffset: Int) {
        buffer.overwriteInt(skipOffset, buffer.size() - (skipOffset + 4))
    }

    /**
     * Starts a `PatternDefine` (`246`) container with a placeholder body byte-length field and
     * returns the buffer offset to pass to [endPatternDefine].
     */
    public fun beginPatternDefine(id: Int, paramIds: IntArray): Int {
        validateOpcode(PatternDefineNode.OPCODE, "PatternDefine")
        buffer.start(PatternDefineNode.OPCODE)
        buffer.writeInt(id)
        buffer.writeInt(paramIds.size)
        for (item in paramIds) {
            buffer.writeInt(item)
        }
        val skipOffset = buffer.index
        buffer.writeInt(0)
        return skipOffset
    }

    /**
     * Backpatches the pattern body byte length at [skipOffset] opened by [beginPatternDefine] and
     * emits the closing `ContainerEnd` (`214`) operation.
     */
    public fun endPatternDefine(skipOffset: Int) {
        buffer.overwriteInt(skipOffset, buffer.size() - (skipOffset + 4))
        containerEnd()
    }

    override fun drawCircle(
        centerX: MaybeVariableFloat,
        centerY: MaybeVariableFloat,
        radius: MaybeVariableFloat,
    ) {
        validateOpcode(DrawCircleNode.OPCODE, "DrawCircle")

    }

    pub fn width(&self) -> i32 {
        ((self.width_and_type as u32) & 0xFFFF) as i32
    }

    pub fn image_type(&self) -> i32 {
        (((self.width_and_type as u32) >> 16) & 0xFFFF) as i32
    }

    pub fn height(&self) -> i32 {
        ((self.height_and_encoding as u32) & 0xFFFF) as i32
    }

    pub fn bitmap_encoding(&self) -> i32 {
        (((self.height_and_encoding as u32) >> 16) & 0xFFFF) as i32
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        buffer.write_i32(self.image_id);
        buffer.write_i32(self.width_and_type);
        buffer.write_i32(self.height_and_encoding);
        buffer.write_buffer(&self.bitmap);
        Ok(())
    }
}

/// Embeds font glyph layout, character metrics, and glyph page mappings for bitmap font rendering
#[derive(Debug, Clone, PartialEq)]
pub struct BitmapFontData {
    /// Bitmap font ID
    pub id: i32,
    /// Glyph definitions
    pub glyphs: Vec<Glyph>,
    /// Optional kerning adjustment pairs (present when version >= 1)
    pub kerning_table: Vec<KerningPair>,
}

impl BitmapFontData {
    /// Wire protocol opcode identifier.
    pub const OP_CODE: u8 = 167;

    /// Reads and decodes a [BitmapFontData] operation from the wire buffer.
    pub fn read(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        let id = buffer.read_i32()?;
        let raw_version_and_num_glyphs = buffer.read_i32()?;
        let num_glyphs = ((raw_version_and_num_glyphs as u32) & 0xFFFF) as i32;
        let version = (raw_version_and_num_glyphs >> 16);
        if num_glyphs < 0 {
            return Err(WireError::InvalidData);
        }
        let glyphs_len = num_glyphs as usize;
        if glyphs_len > buffer.remaining() / 20 {
            return Err(WireError::Underflow);
        }
        let mut glyphs_vec = Vec::with_capacity(glyphs_len.min(4096));
        for _ in 0..glyphs_len {
            glyphs_vec.push(Glyph::read(buffer)?);
        }
        let glyphs = glyphs_vec;
        let kerning_table = if version >= 1 {
            let kerning_table_len = (buffer.read_i16()? as u16) as usize;
            if kerning_table_len > buffer.remaining() / 6 {
                return Err(WireError::Underflow);
            }
            let mut kerning_table_vec = Vec::with_capacity(kerning_table_len.min(4096));
            for _ in 0..kerning_table_len {
                kerning_table_vec.push(KerningPair::read(buffer)?);
            }
            kerning_table_vec
        } else {
            Vec::new()
        };
        Ok(Self {
            id,
            glyphs,
            kerning_table,
        })
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        buffer.write_i32(self.id);
        let num_glyphs: i32 = self.glyphs.len() as i32;
        let version: i32 = if !self.kerning_table.is_empty() { 1 } else { 0 };
        buffer.write_i32((((num_glyphs as u32) & 0xFFFF) | (((version as u32) & 0xFFFF) << 16)) as i32);
        for item in &self.glyphs {
            item.write(buffer)?;
        }
        if version >= 1 {
            buffer.write_i16(self.kerning_table.len() as i16);
            for item in &self.kerning_table {
                item.write(buffer)?;
            }
        }
        Ok(())
    }
}

/// Extract time-related information (seconds, hours, etc.)
#[derive(Debug, Clone, PartialEq)]
pub struct TimeAttribute {
    /// The ID of the float variable to store the result
    pub id: i32,
    /// The ID of the time variable to extract from
    pub time_id: i32,
    /// The type of time information to extract
    pub r#type: i16,
    /// The optional arguments
    pub args: Vec<i32>,
}

impl TimeAttribute {
    /// Wire protocol opcode identifier.
    pub const OP_CODE: u8 = 172;

    /// Reads and decodes a [TimeAttribute] operation from the wire buffer.
    pub fn read(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        let id = buffer.read_i32()?;
        let time_id = buffer.read_i32()?;
        let r#type = buffer.read_i16()?;
        let args_raw_len = buffer.read_i16()?;
        if args_raw_len < 0 {
            return Err(WireError::InvalidData);
        }
        let args_len = args_raw_len as usize;
        if args_len > 32 {
            return Err(WireError::InvalidData);
        }
        if args_len > buffer.remaining() / 4 {
            return Err(WireError::Underflow);
        }
        let mut args_vec = Vec::with_capacity(args_len.min(4096));
        for _ in 0..args_len {
            args_vec.push(buffer.read_i32()?);
        }
        let args = args_vec;
        Ok(Self {
            id,
            time_id,
            r#type,
            args,
        })
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        buffer.write_i32(self.id);
        buffer.write_i32(self.time_id);
        buffer.write_i16(self.r#type);
        buffer.write_i16(self.args.len() as i16);
        for &item in &self.args {
            buffer.write_i32(item);
        }
        Ok(())
    }
}

/// Execute a list of operations if a condition is met
#[derive(Debug, Clone, PartialEq)]
pub struct ConditionalOperations {
    /// The type of comparison (EQ, NEQ, LT, etc.)
    pub r#type: u8,
    /// The first value to compare
    pub var_a: f32,
    /// The second value to compare
    pub var_b: f32,
}

impl ConditionalOperations {
    /// Wire protocol opcode identifier.
    pub const OP_CODE: u8 = 178;

    /// Reads and decodes a [ConditionalOperations] operation from the wire buffer.
    pub fn read(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        let r#type = buffer.read_u8()?;
        if !matches!(r#type, 0 | 1 | 2 | 3 | 4 | 5 | 6) {
            return Err(WireError::InvalidData);
        }
        let var_a = buffer.read_nan_id()?;
        let var_b = buffer.read_nan_id()?;
        Ok(Self {
            r#type,
            var_a,
            var_b,
        })
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        buffer.write_u8(self.r#type);
        buffer.write_nan_id(self.var_a);
        buffer.write_nan_id(self.var_b);
        Ok(())
    }
}

/// Measure text dimensions specifically for bitmap fonts
#[cfg(any(feature = "profile_all", rc_profile_all, any(feature = "profile_androidx", rc_profile_androidx, feature = "profile_widgets", rc_profile_widgets), not(any(feature = "profile_baseline", feature = "profile_widgets", feature = "profile_androidx", feature = "profile_experimental", feature = "profile_deprecated", feature = "profile_oem", rc_profile_baseline, rc_profile_widgets, rc_profile_androidx, rc_profile_experimental, rc_profile_deprecated, rc_profile_oem))))]
#[derive(Debug, Clone, PartialEq)]
pub struct BitmapTextMeasure {
    /// Output float ID in bits 0..30 and glyphSpacing presence flag in bit 31
    pub id: i32,
    /// Optional horizontal spacing between glyphs in pixels
    pub glyph_spacing: Option<f32>,
    /// The ID of the text to measure
    pub text_id: i32,
    /// The ID of the bitmap font
    pub bitmap_font_id: i32,
    /// Measurement type and flags
    pub r#type: i32,
}

#[cfg(any(feature = "profile_all", rc_profile_all, any(feature = "profile_androidx", rc_profile_androidx, feature = "profile_widgets", rc_profile_widgets), not(any(feature = "profile_baseline", feature = "profile_widgets", feature = "profile_androidx", feature = "profile_experimental", feature = "profile_deprecated", feature = "profile_oem", rc_profile_baseline, rc_profile_widgets, rc_profile_androidx, rc_profile_experimental, rc_profile_deprecated, rc_profile_oem))))]
impl BitmapTextMeasure {
    /// Wire protocol opcode identifier.
    pub const OP_CODE: u8 = 183;

    /// Reads and decodes a [BitmapTextMeasure] operation from the wire buffer.
    pub fn read(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        let raw_id = buffer.read_i32()?;
        let id = ((raw_id as u32) & 0x7FFFFFFF) as i32;
        let has_glyph_spacing = (((raw_id as u32) >> 31) & 0x1) as i32;
        let glyph_spacing = if has_glyph_spacing != 0 {
            Some(buffer.read_nan_id()?)
        } else {
            None
        };
        let text_id = buffer.read_i32()?;
        let bitmap_font_id = buffer.read_i32()?;
        let r#type = buffer.read_i32()?;
        Ok(Self {
            id,
            glyph_spacing,
            text_id,
            bitmap_font_id,
            r#type,
        })
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        let has_glyph_spacing: i32 = if self.glyph_spacing.is_some() { 1 } else { 0 };
        buffer.write_i32((((self.id as u32) & 0x7FFFFFFF) | (((has_glyph_spacing as u32) & 0x1) << 31)) as i32);
        if has_glyph_spacing != 0 {
            let val = self.glyph_spacing.as_ref().ok_or(WireError::MissingRequiredField("glyphSpacing"))?;
            buffer.write_nan_id(*val);
        }
        buffer.write_i32(self.text_id);
        buffer.write_i32(self.bitmap_font_id);
        buffer.write_i32(self.r#type);
        Ok(())
    }
}

/// Text style implementation
#[cfg(any(feature = "profile_all", rc_profile_all, any(feature = "profile_androidx", rc_profile_androidx, feature = "profile_widgets", rc_profile_widgets), not(any(feature = "profile_baseline", feature = "profile_widgets", feature = "profile_androidx", feature = "profile_experimental", feature = "profile_deprecated", feature = "profile_oem", rc_profile_baseline, rc_profile_widgets, rc_profile_androidx, rc_profile_experimental, rc_profile_deprecated, rc_profile_oem))))]
#[derive(Debug, Clone, PartialEq)]
pub struct TextStyle {
    pub id: Option<i32>,
    pub font_size: Option<f32>,
    pub font_axis: Option<Vec<i32>>,
}

#[cfg(any(feature = "profile_all", rc_profile_all, any(feature = "profile_androidx", rc_profile_androidx, feature = "profile_widgets", rc_profile_widgets), not(any(feature = "profile_baseline", feature = "profile_widgets", feature = "profile_androidx", feature = "profile_experimental", feature = "profile_deprecated", feature = "profile_oem", rc_profile_baseline, rc_profile_widgets, rc_profile_androidx, rc_profile_experimental, rc_profile_deprecated, rc_profile_oem))))]
impl TextStyle {
    /// Wire protocol opcode identifier.
    pub const OP_CODE: u8 = 242;

    /// Reads and decodes a [TextStyle] operation from the wire buffer.
    pub fn read(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        let mut id: Option<i32> = None;
        let mut font_size: Option<f32> = None;
        let mut font_axis: Option<Vec<i32>> = None;
        let sparse_count = (buffer.read_i16()? as u16) as usize;
        if sparse_count > buffer.remaining() / 2 {
            return Err(WireError::Underflow);
        }
        for _ in 0..sparse_count {
            let param_id = buffer.read_u8()?;
            match param_id {
                1 => { id = Some(buffer.read_i32()?); }
                5 => { font_size = Some(buffer.read_nan_id()?); }
                20 => {
                    let font_axis_len = (buffer.read_i16()? as u16) as usize;
                    if font_axis_len > buffer.remaining() / 4 {
                        return Err(WireError::Underflow);
                    }
                    let mut font_axis_vec = Vec::with_capacity(font_axis_len.min(4096));
                    for _ in 0..font_axis_len {
                        font_axis_vec.push(buffer.read_i32()?);
                    }
                    font_axis = Some(font_axis_vec);
                }
                _ => return Err(WireError::InvalidData),
            }
        }
        Ok(Self {
            id,
            font_size,
            font_axis,
        })
    }

    #[cfg(any(feature = "writer", not(any(rc_no_writer, feature = "reader_only"))))]
    /// Encodes and writes this operation to the wire buffer.
    pub fn write(&self, buffer: &mut WireBuffer) -> Result<(), WireError> {
        buffer.start(Self::OP_CODE);
        let mut sparse_count: usize = 0;
        if self.id.is_some() { sparse_count += 1; }
        if self.font_size.is_some() { sparse_count += 1; }
        if self.font_axis.is_some() { sparse_count += 1; }
        buffer.write_i16(sparse_count as i16);
        if let Some(ref val) = self.id {
            buffer.write_u8(1);
            buffer.write_i32(*val);
        }
        if let Some(ref val) = self.font_size {
            buffer.write_u8(5);
            buffer.write_nan_id(*val);
        }
        if let Some(ref val) = self.font_axis {
            buffer.write_u8(20);
            buffer.write_i16(val.len() as i16);
            for &item in val {
                buffer.write_i32(item);
            }
        }
        Ok(())
    }
}

/// Top-level RemoteCompose document container in Rust.
#[derive(Debug, Clone, PartialEq)]
pub struct RemoteDocument {
    pub major_version: i32,
    pub minor_version: i32,
    pub patch_version: i32,
    pub width: i32,
    pub height: i32,
    pub capabilities: i64,
    pub operations: Vec<Operation>,
}

impl Default for RemoteDocument {
    fn default() -> Self {
        Self::new()
    }
}

impl RemoteDocument {
    pub const MAX_HEADER_PROPERTIES: usize = 1000;
    pub const MAX_OPERATION_COUNT: usize = 100_000;

    /// Creates a new, empty [RemoteDocument].
    pub fn new() -> Self {
        Self {
            major_version: 1,
            minor_version: 1,
            patch_version: 0,
            width: 0,
            height: 0,
            capabilities: 0,
            operations: Vec::new(),
        }
    }

    /// Loads and deserializes a [RemoteDocument] from the wire buffer with default limits.
    pub fn load(buffer: &mut WireBuffer) -> Result<Self, WireError> {
        Self::load_with_limits(buffer, Self::MAX_OPERATION_COUNT)
    }

    /// Loads and deserializes a [RemoteDocument] evaluating conditional skip operations
    /// against the target player API level and profile bitmask.
    pub fn load_for_player(
        buffer: &mut WireBuffer,
        player_api_level: i32,
        player_profiles: i32,
    ) -> Result<Self, WireError> {
        Self::load_for_player_with_limits(
            buffer,
            player_api_level,
            player_profiles,
            Self::MAX_OPERATION_COUNT,
        )
    }

    /// Loads and deserializes a [RemoteDocument] evaluating conditional skip operations
    /// with an explicit maximum operation count.
    pub fn load_for_player_with_limits(
        buffer: &mut WireBuffer,
        player_api_level: i32,
        player_profiles: i32,
        max_operations: usize,
    ) -> Result<Self, WireError> {
        Self::load_impl(buffer, max_operations, Some((player_api_level, player_profiles)))
    }

    /// Loads and deserializes a [RemoteDocument] from the wire buffer with an explicit
    /// maximum operation count.
    pub fn load_with_limits(
        buffer: &mut WireBuffer,
        max_operations: usize,
    ) -> Result<Self, WireError> {
        Self::load_impl(buffer, max_operations, None)
    }

    fn load_impl(
        buffer: &mut WireBuffer,
        max_operations: usize,
        player_ctx: Option<(i32, i32)>,
    ) -> Result<Self, WireError> {
        let _ = player_ctx;
        let mut doc = Self::new();
        if !buffer.available() {
            return Ok(doc);

  static readonly MAX_OPERATION_COUNT = 100000;

  height: number = 0;

  operations: Operation[] = [];

  static load(
    bytes: Uint8Array,
    maxOperations: number = RemoteDocument.MAX_OPERATION_COUNT,
  ): RemoteDocument {
    return RemoteDocument.loadImpl(bytes, maxOperations, false, 0, 0);
  }

  /**
   * Loads and parses a RemoteDocument evaluating conditional skip operations
   * against the target player API level and profile bitmask.
   */
  static loadForPlayer(
    bytes: Uint8Array,
    playerApiLevel: number,
    playerProfiles: number = 0,
    maxOperations: number = RemoteDocument.MAX_OPERATION_COUNT,
  ): RemoteDocument {
    return RemoteDocument.loadImpl(
      bytes,
      maxOperations,
      true,
      playerApiLevel,
      playerProfiles,
    );
  }

  private static loadImpl(
    bytes: Uint8Array,
    maxOperations: number,
    evaluateSkip: boolean,
    playerApiLevel: number,
    playerProfiles: number,
  ): RemoteDocument {
    const buffer = WireBuffer.fromBytes(bytes);
    const doc = new RemoteDocument();
    if (!buffer.available()) return doc;

    const headerOp = buffer.readByte();
    if (headerOp !== OpCode.Header) {
      throw new Error("Expected Header opcode 0, got " + headerOp);
    }
    const rawMajor = buffer.readInt();
    doc.minorVersion = buffer.readInt();
    doc.patchVersion = buffer.readInt();
    if ((rawMajor & 0xFFFF0000) === 0x048C0000) {
      doc.majorVersion = rawMajor & 0xFFFF;
      doc.width = 256;
      doc.height = 256;
      const numProps = buffer.readInt();
      if (numProps < 0 || numProps > RemoteDocument.MAX_HEADER_PROPERTIES) {
        throw new Error("Invalid numProps in header: " + numProps);
      }
      if (numProps * 4 > buffer.remaining()) {
        throw new RangeError(
          "Header property count exceeds remaining WireBuffer size: " + numProps
        );
      }
      for (let i = 0; i < numProps; i++) {
        const tag = buffer.readShort();
        buffer.readShort(); // itemLen (read and ignored)
        const dataType = tag >> 10;
        const key = tag & 0x3F;
        switch (dataType) {
          case 0: {
            const val = buffer.readInt();
            if (key === 5) doc.width = val;
            else if (key === 6) doc.height = val;
            else if (key === 14) doc.capabilities = BigInt(val >>> 0);
            else if (key === 31 && val !== 0) {
              throw new Error("Compressed RemoteCompose documents are not supported");
            }
            break;
          }
          case 1: buffer.readFloat(); break;
          case 2: {
            const val = buffer.readLong();
            if (key === 14) doc.capabilities = val;
            else if (key === 31 && val !== 0n) {
              throw new Error("Compressed RemoteCompose documents are not supported");
            }
            break;
          }
          case 3: buffer.readUTF8(4000); break;
          default:
            break;
        }
      }
    } else if (rawMajor >= 0 && rawMajor < 0x10000) {
      doc.majorVersion = rawMajor;
      doc.width = buffer.readInt();
      doc.height = buffer.readInt();
      doc.capabilities = buffer.readLong();
    } else {
      throw new Error("Invalid header magic number: " + rawMajor);
    }

    while (buffer.available()) {
      const opByte = buffer.readByte();
      if (opByte === OpCode.Header) {
        throw new Error("Unexpected Header opcode in document body");
      }
      if (doc.operations.length >= maxOperations) {
        throw new Error("Document exceeds maximum allowed operation count: " + maxOperations);
      }
      const op = createOperation(opByte as OpCode);
      if (!op) {
        throw new Error("Unknown opcode in RemoteDocument: " + opByte);
      }
      if (evaluateSkip && typeof (op as any).readFieldsForPlayer === "function") {
        (op as any).readFieldsForPlayer(buffer, playerApiLevel, playerProfiles);
      } else {
        (op as any).readFields(buffer);
      }
      doc.operations.push(op);
    }
    return doc;
  }

  /** Serializes this document to binary bytes. */
  save(): Uint8Array {
    const buffer = new WireBuffer();
    buffer.writeByte(OpCode.Header);
    buffer.writeInt(this.majorVersion);
    buffer.writeInt(this.minorVersion);
    buffer.writeInt(this.patchVersion);
    buffer.writeInt(this.width);
    buffer.writeInt(this.height);
    buffer.writeLong(this.capabilities);

    for (const op of this.operations) {
      op.write(buffer);
    }
    return buffer.getBytes();
  }
}

    public init() {}

        return try loadImpl(from: data, maxOperations: maxOperations, evaluateSkip: false, playerApiLevel: 0, playerProfiles: 0)

    /// Deserializes a RemoteDocument evaluating conditional skip operations against the target player.
    public static func loadForPlayer(
        from data: Data,
        playerApiLevel: Int32,
        playerProfiles: Int32 = 0,
        maxOperations: Int = MAX_OPERATION_COUNT
    ) throws -> RemoteDocument {

            from: data,
            maxOperations: maxOperations,
            evaluateSkip: true,
            playerApiLevel: playerApiLevel,
            playerProfiles: playerProfiles
        )
    }

    private static func loadImpl(
        from data: Data,
        maxOperations: Int,
        evaluateSkip: Bool,
        playerApiLevel: Int32,
        playerProfiles: Int32
    ) throws -> RemoteDocument {
        var buffer = WireBuffer(data: data)
        var doc = RemoteDocument()
        guard buffer.available else { return doc }

        let headerOp = try buffer.readByte()
        guard headerOp == OpCode.header.rawValue else {
            throw NSError(domain: "RemoteDocument", code: 2, userInfo: [NSLocalizedDescriptionKey: "Expected header opcode"])
        }
        let rawMajor = try buffer.readInt()
        doc.minorVersion = try buffer.readInt()
        doc.patchVersion = try buffer.readInt()
        if (UInt32(bitPattern: rawMajor) & 0xFFFF0000) == 0x048C0000 {
            doc.majorVersion = rawMajor & 0xFFFF
            doc.width = 256
            doc.height = 256
            let numProps = Int(try buffer.readInt())
            guard numProps >= 0 && numProps <= MAX_HEADER_PROPERTIES else {
                throw NSError(domain: "RemoteDocument", code: 3, userInfo: [NSLocalizedDescriptionKey: "Invalid header property count: \(numProps)"])
            }
            guard numProps <= buffer.remaining / 4 else {
                throw NSError(domain: "RemoteDocument", code: 3, userInfo: [NSLocalizedDescriptionKey: "Header property count exceeds remaining buffer"])
            }
            for _ in 0..<numProps {
                let tag = Int32(try buffer.readShort())
                _ = try buffer.readShort() // itemLen (read and ignored)
                let dataType = tag >> 10
                let key = tag & 0x3F
                switch dataType {
                case 0:
                    let val = try buffer.readInt()
                    if key == 5 { doc.width = val }
                    else if key == 6 { doc.height = val }
                    else if key == 14 { doc.capabilities = Int64(UInt32(bitPattern: val)) }
                    else if key == 31 && val != 0 {
                        throw NSError(domain: "RemoteDocument", code: 6, userInfo: [NSLocalizedDescriptionKey: "Compressed RemoteCompose documents are not supported"])
                    }
                case 1:
                    _ = try buffer.readFloat()
                case 2:
                    let val = try buffer.readLong()
                    if key == 14 { doc.capabilities = val }
                    else if key == 31 && val != 0 {
                        throw NSError(domain: "RemoteDocument", code: 6, userInfo: [NSLocalizedDescriptionKey: "Compressed RemoteCompose documents are not supported"])
                    }
                case 3:
                    _ = try buffer.readUTF8(maxLength: 4000)
                default:
                    break
                }
            }
        } else if rawMajor >= 0 && rawMajor < 0x10000 {
            doc.majorVersion = rawMajor
            doc.width = try buffer.readInt()
            doc.height = try buffer.readInt()
            doc.capabilities = try buffer.readLong()
        } else {
            throw NSError(domain: "RemoteDocument", code: 7, userInfo: [NSLocalizedDescriptionKey: "Invalid header magic number: \(rawMajor)"])
        }

        while buffer.available {
            guard doc.operations.count < maxOperations else {
                throw NSError(domain: "RemoteDocument", code: 8, userInfo: [NSLocalizedDescriptionKey: "Operation count exceeds limit \(maxOperations)"])
            }
            let opByte = try buffer.readByte()
            guard let opCode = OpCode(rawValue: opByte) else {
                throw NSError(domain: "RemoteDocument", code: 5, userInfo: [NSLocalizedDescriptionKey: "Unknown opcode: \(opByte)"])
            }
            switch opCode {

                throw NSError(domain: "RemoteDocument", code: 2, userInfo: [NSLocalizedDescriptionKey: "Unexpected Header opcode inside document body"])

                doc.operations.append(try DrawCircle.read(from: &buffer))
            case .floatExpression:
                doc.operations.append(try FloatExpression.read(from: &buffer))
            case .bitmapData:
                doc.operations.append(try BitmapData.read(from: &buffer))
            case .textData:
                doc.operations.append(try TextData.read(from: &buffer))
            case .bitmapFontData:
                doc.operations.append(try BitmapFontData.read(from: &buffer))
            case .timeAttribute:
                doc.operations.append(try TimeAttribute.read(from: &buffer))
            case .conditionalOperations:
                doc.operations.append(try ConditionalOperations.read(from: &buffer))
            case .bitmapTextMeasure:
                doc.operations.append(try BitmapTextMeasure.read(from: &buffer))
            case .containerEnd:
                doc.operations.append(try ContainerEnd.read(from: &buffer))
            case .textStyle:
                doc.operations.append(try TextStyle.read(from: &buffer))
            }
        }
        return doc
    }

    /// Serializes this document to raw binary data.
    public func save() throws -> Data {
        var buffer = WireBuffer()
        buffer.writeByte(OpCode.header.rawValue)
        buffer.writeInt(majorVersion)
        buffer.writeInt(minorVersion)
        buffer.writeInt(patchVersion)

        buffer.writeInt(height)

        }

    }

from typing import ClassVar, Dict, List, Optional, Type


class OpCode(enum.IntEnum):
    """Operation wire identifiers for RemoteCompose."""
    # Document metadata containing version, dimensions, density behavior, and capabilities
    Header = 0
    # Draws a circle with center (centerX, centerY) and radius
    DrawCircle = 46
    # Defines a float variable computed via Reverse Polish Notation (RPN) expression evaluation
    FloatExpression = 81
    # Embed or reference bitmap image data
    BitmapData = 101
    # Define a static string and associate it with an ID
    TextData = 102
    # Embeds font glyph layout, character metrics, and glyph page mappings for bitmap font rendering
    BitmapFontData = 167
    # Extract time-related information (seconds, hours, etc.)
    TimeAttribute = 172
    # Execute a list of operations if a condition is met
    ConditionalOperations = 178
    # Measure text dimensions specifically for bitmap fonts
    BitmapTextMeasure = 183
    # Closes the current container or layout component
    ContainerEnd = 214
    # Text style implementation
    TextStyle = 242


class Operation:
    """Base class for all RemoteCompose operations."""
    op_code: ClassVar[OpCode]

    def write(self, buffer: WireBuffer) -> None:
        raise NotImplementedError

    def read_fields(self, buffer: WireBuffer) -> None:
        raise NotImplementedError


class WireBuffer:
    """Big-endian binary buffer for reading and writing RemoteCompose wire format."""

    def __init__(self, data: Optional[bytes] = None):
        if data is not None:
            self._buffer = bytearray(data)
            self._index = 0
            self._size = len(data)
        else:
            self._buffer = bytearray(1024)
            self._index = 0
            self._size = 0

    @classmethod
    def from_bytes(cls, data: bytes) -> WireBuffer:
        return cls(data)

    def get_bytes(self) -> bytes:
        return bytes(self._buffer[:self._size])

    def available(self) -> bool:
        return self._index < self._size

    def remaining(self) -> int:
        return max(0, self._size - self._index)

    def skip_bytes(self, length: int) -> None:
        if length < 0:
            raise ValueError(f"Negative skip length in WireBuffer: {length}")
        if length > self.remaining():
            raise EOFError("WireBuffer underflow skipping bytes")
        self._index += length

    @property
    def position(self) -> int:
        return self._index

    @position.setter
    def position(self, pos: int) -> None:
        if pos < 0 or pos > self._size:
            raise ValueError(f"WireBuffer position out of bounds: {pos} (size={self._size})")
        self._index = pos

    @property
    def size(self) -> int:
        return self._size

    def _ensure_capacity(self, needed: int) -> None:
        required = self._index + needed
        if required > len(self._buffer):
            new_capacity = max(len(self._buffer) * 2, required)
            self._buffer.extend(b"\x00" * (new_capacity - len(self._buffer)))

    def write_byte(self, val: int) -> None:
        self._ensure_capacity(1)
        struct.pack_into(">B", self._buffer, self._index, val & 0xFF)
        self._index += 1
        if self._index > self._size:
            self._size = self._index

    def read_byte(self) -> int:
        if 1 > self.remaining():
            raise EOFError("WireBuffer underflow reading byte")
        val = struct.unpack_from(">B", self._buffer, self._index)[0]
        self._index += 1
        return val

    def write_short(self, val: int) -> None:
        self._ensure_capacity(2)
        v = val & 0xFFFF
        if v >= 0x8000:
            v -= 0x10000
        struct.pack_into(">h", self._buffer, self._index, v)
        self._index += 2
        if self._index > self._size:
            self._size = self._index

    def read_short(self) -> int:
        if 2 > self.remaining():
            raise EOFError("WireBuffer underflow reading short")
        val = struct.unpack_from(">h", self._buffer, self._index)[0]
        self._index += 2
        return val

    def write_int(self, val: int) -> None:
        self._ensure_capacity(4)
        v = val & 0xFFFFFFFF
        if v >= 0x80000000:
            v -= 0x100000000
        struct.pack_into(">i", self._buffer, self._index, v)
        self._index += 4
        if self._index > self._size:
            self._size = self._index

    def read_int(self) -> int:
        if 4 > self.remaining():
            raise EOFError("WireBuffer underflow reading int")
        val = struct.unpack_from(">i", self._buffer, self._index)[0]
        self._index += 4
        return val

    def write_long(self, val: int) -> None:
        self._ensure_capacity(8)
        struct.pack_into(">q", self._buffer, self._index, val)
        self._index += 8
        if self._index > self._size:
            self._size = self._index

    def read_long(self) -> int:
        if 8 > self.remaining():
            raise EOFError("WireBuffer underflow reading long")
        val = struct.unpack_from(">q", self._buffer, self._index)[0]
        self._index += 8
        return val

    def write_float(self, val: float) -> None:
        self._ensure_capacity(4)
        struct.pack_into(">f", self._buffer, self._index, val)
        self._index += 4
        if self._index > self._size:
            self._size = self._index

    def read_float(self) -> float:
        if 4 > self.remaining():
            raise EOFError("WireBuffer underflow reading float")
        val = struct.unpack_from(">f", self._buffer, self._index)[0]
        self._index += 4
        return val

    def write_nan_id(self, val: float) -> None:
        self.write_float(val)

    def read_nan_id(self) -> float:
        return self.read_float()

    def write_boolean(self, val: bool) -> None:
        self.write_byte(1 if val else 0)

    def read_boolean(self) -> bool:
        return self.read_byte() != 0

    def write_utf8(self, val: str) -> None:
        encoded = val.encode("utf-8")
        self.write_int(len(encoded))
        self._ensure_capacity(len(encoded))
        self._buffer[self._index : self._index + len(encoded)] = encoded
        self._index += len(encoded)
        if self._index > self._size:
            self._size = self._index

    def read_utf8(self, max_length: Optional[int] = None) -> str:
        length = self.read_int()
        if length < 0:
            raise ValueError(f"Negative string length in WireBuffer: {length}")
        if max_length is not None and length > max_length:
            raise ValueError(f"String length {length} exceeds maximum ({max_length})")
        if length > self.remaining():
            raise EOFError("WireBuffer underflow reading utf8 string")
        raw = self._buffer[self._index : self._index + length]
        decoded = raw.decode("utf-8")
        self._index += length
        return decoded

    def write_buffer(self, val: bytes) -> None:
        self.write_int(len(val))
        self._ensure_capacity(len(val))
        self._buffer[self._index : self._index + len(val)] = val
        self._index += len(val)
        if self._index > self._size:
            self._size = self._index

    def write_bytes(self, val: bytes) -> None:
        self._ensure_capacity(len(val))
        self._buffer[self._index : self._index + len(val)] = val
        self._index += len(val)
        if self._index > self._size:
            self._size = self._index

    def read_bytes(self, length: int) -> bytes:
        if length < 0:
            raise ValueError(f"Negative byte slice length in WireBuffer: {length}")
        if length > self.remaining():
            raise EOFError("WireBuffer underflow reading raw bytes")
        raw = bytes(self._buffer[self._index : self._index + length])
        self._index += length
        return raw

    def read_buffer(self, max_length: Optional[int] = None) -> bytes:
        length = self.read_int()
        if length < 0:
            raise ValueError(f"Negative buffer length in WireBuffer: {length}")
        if max_length is not None and length > max_length:
            raise ValueError(f"Buffer length {length} exceeds maximum ({max_length})")
        if length > self.remaining():
            raise EOFError("WireBuffer underflow reading buffer")
        raw = bytes(self._buffer[self._index : self._index + length])
        self._index += length
        return raw


@dataclass
class Glyph:
    """Bitmap font glyph metrics and bitmap reference"""
    character: str = ""
    bitmap_id: int = 0
    margin_left: int = 0
    margin_top: int = 0
    margin_right: int = 0
    margin_bottom: int = 0
    width: int = 0
    height: int = 0

    def read_fields(self, buffer: WireBuffer) -> None:
        self.character = buffer.read_utf8()
        self.bitmap_id = buffer.read_int()
        self.margin_left = buffer.read_short()
        self.margin_top = buffer.read_short()
        self.margin_right = buffer.read_short()
        self.margin_bottom = buffer.read_short()
        self.width = buffer.read_short()
        self.height = buffer.read_short()

    @classmethod
    def read(cls, buffer: WireBuffer) -> Glyph:
        inst = cls()
        inst.read_fields(buffer)
        return inst

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_utf8(self.character)
        buffer.write_int(self.bitmap_id)
        buffer.write_short(self.margin_left)
        buffer.write_short(self.margin_top)
        buffer.write_short(self.margin_right)
        buffer.write_short(self.margin_bottom)
        buffer.write_short(self.width)
        buffer.write_short(self.height)


@dataclass
class KerningPair:
    """Kerning adjustment for a pair of glyphs"""
    glyph_pair: str = ""
    adjustment: int = 0

    def read_fields(self, buffer: WireBuffer) -> None:
        self.glyph_pair = buffer.read_utf8()
        self.adjustment = buffer.read_short()

    @classmethod
    def read(cls, buffer: WireBuffer) -> KerningPair:
        inst = cls()
        inst.read_fields(buffer)
        return inst

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_utf8(self.glyph_pair)
        buffer.write_short(self.adjustment)


@dataclass
class Header(Operation):
    """Document metadata containing version, dimensions, density behavior, and capabilities"""
    op_code: ClassVar[OpCode] = OpCode.Header
    major_version: int = 0
    minor_version: int = 0
    patch_version: int = 0
    width: int = 0
    height: int = 0
    capabilities: int = 0

    def read_fields(self, buffer: WireBuffer) -> None:
        self.major_version = buffer.read_int()
        self.minor_version = buffer.read_int()
        self.patch_version = buffer.read_int()
        self.width = buffer.read_int()
        self.height = buffer.read_int()
        self.capabilities = buffer.read_long()

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.major_version)
        buffer.write_int(self.minor_version)
        buffer.write_int(self.patch_version)
        buffer.write_int(self.width)
        buffer.write_int(self.height)
        buffer.write_long(self.capabilities)

@dataclass
class FloatExpression(Operation):
    """Defines a float variable computed via Reverse Polish Notation (RPN) expression evaluation"""
    op_code: ClassVar[OpCode] = OpCode.FloatExpression
    id: int = 0
    expression: List[float] = field(default_factory=list)
    animation_spec: List[float] = field(default_factory=list)

    def read_fields(self, buffer: WireBuffer) -> None:
        self.id = buffer.read_int()
        _raw_packed_lengths = buffer.read_int()
        expression_length = (_raw_packed_lengths & 0xFFFF)
        animation_length = (((_raw_packed_lengths & 0xFFFFFFFF) >> 16) & 0xFFFF)
        _expression_count = expression_length
        if _expression_count < 0:
            raise ValueError(f"Invalid negative array length for expression: {_expression_count}")
        if _expression_count > 32:
            raise ValueError(f"Array length {_expression_count} for expression exceeds maximum (32)")
        if _expression_count > buffer.remaining() // 4:
            raise EOFError(f"Array length {_expression_count} for expression exceeds remaining buffer ({buffer.remaining()})")
        self.expression = [buffer.read_nan_id() for _ in range(_expression_count)]
        _animation_spec_count = animation_length
        if _animation_spec_count < 0:
            raise ValueError(f"Invalid negative array length for animation_spec: {_animation_spec_count}")
        if _animation_spec_count > buffer.remaining() // 4:
            raise EOFError(f"Array length {_animation_spec_count} for animation_spec exceeds remaining buffer ({buffer.remaining()})")
        self.animation_spec = [buffer.read_nan_id() for _ in range(_animation_spec_count)]

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.id)
        expression_length = len(self.expression)
        animation_length = len(self.animation_spec)
        buffer.write_int((((((expression_length & 0xFFFF)) | (((animation_length & 0xFFFF) << 16))) & 0xFFFFFFFF) - 0x100000000 if ((((expression_length & 0xFFFF)) | (((animation_length & 0xFFFF) << 16))) & 0xFFFFFFFF) >= 0x80000000 else ((((expression_length & 0xFFFF)) | (((animation_length & 0xFFFF) << 16))) & 0xFFFFFFFF)))
        for item in self.expression:
            buffer.write_nan_id(item)
        for item in self.animation_spec:
            buffer.write_nan_id(item)


@dataclass
class BitmapData(Operation):
    """Embed or reference bitmap image data"""
    op_code: ClassVar[OpCode] = OpCode.BitmapData
    image_id: int = 0
    width_and_type: int = 0
    height_and_encoding: int = 0
    bitmap: bytes = b""

    def read_fields(self, buffer: WireBuffer) -> None:
        self.image_id = buffer.read_int()
        self.width_and_type = buffer.read_int()
        self.height_and_encoding = buffer.read_int()
        self.bitmap = buffer.read_buffer()

    @property
    def width(self) -> int:
        return (self.width_and_type & 0xFFFF)

    @property
    def image_type(self) -> int:
        return (((self.width_and_type & 0xFFFFFFFF) >> 16) & 0xFFFF)

    @property
    def height(self) -> int:
        return (self.height_and_encoding & 0xFFFF)

    @property
    def bitmap_encoding(self) -> int:
        return (((self.height_and_encoding & 0xFFFFFFFF) >> 16) & 0xFFFF)

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.image_id)
        buffer.write_int(self.width_and_type)
        buffer.write_int(self.height_and_encoding)
        buffer.write_buffer(self.bitmap)


@dataclass
class TextData(Operation):
    """Define a static string and associate it with an ID"""
    op_code: ClassVar[OpCode] = OpCode.TextData
    text_id: int = 0
    text: str = ""

    def read_fields(self, buffer: WireBuffer) -> None:
        self.text_id = buffer.read_int()
        self.text = buffer.read_utf8(4000)

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.text_id)
        buffer.write_utf8(self.text)


@dataclass
class BitmapFontData(Operation):
    """Embeds font glyph layout, character metrics, and glyph page mappings for bitmap font rendering"""
    op_code: ClassVar[OpCode] = OpCode.BitmapFontData
    id: int = 0
    glyphs: List[Glyph] = field(default_factory=list)
    kerning_table: List[KerningPair] = field(default_factory=list)

    def read_fields(self, buffer: WireBuffer) -> None:
        self.id = buffer.read_int()
        _raw_version_and_num_glyphs = buffer.read_int()
        num_glyphs = (_raw_version_and_num_glyphs & 0xFFFF)
        version = ((((_raw_version_and_num_glyphs & 0xFFFFFFFF) >> 16) & 0xFFFF) - (0x10000 if (((_raw_version_and_num_glyphs & 0xFFFFFFFF) >> 31) & 1) else 0))
        _glyphs_count = num_glyphs
        if _glyphs_count < 0:
            raise ValueError(f"Invalid negative array length for glyphs: {_glyphs_count}")
        if _glyphs_count > buffer.remaining() // 20:
            raise EOFError(f"Array length {_glyphs_count} for glyphs exceeds remaining buffer ({buffer.remaining()})")
        self.glyphs = [Glyph.read(buffer) for _ in range(_glyphs_count)]
        if version >= 1:
            _kerning_table_count = (buffer.read_short() & 0xFFFF)
            if _kerning_table_count < 0:
                raise ValueError(f"Invalid negative array length for kerning_table: {_kerning_table_count}")
            if _kerning_table_count > buffer.remaining() // 6:
                raise EOFError(f"Array length {_kerning_table_count} for kerning_table exceeds remaining buffer ({buffer.remaining()})")
            self.kerning_table = [KerningPair.read(buffer) for _ in range(_kerning_table_count)]
        else:
            self.kerning_table = []

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.id)
        num_glyphs = len(self.glyphs)
        version = (1 if len(self.kerning_table) > 0 else 0)
        buffer.write_int((((((num_glyphs & 0xFFFF)) | (((version & 0xFFFF) << 16))) & 0xFFFFFFFF) - 0x100000000 if ((((num_glyphs & 0xFFFF)) | (((version & 0xFFFF) << 16))) & 0xFFFFFFFF) >= 0x80000000 else ((((num_glyphs & 0xFFFF)) | (((version & 0xFFFF) << 16))) & 0xFFFFFFFF)))
        for item in self.glyphs:
            item.write(buffer)
        if version >= 1:
            buffer.write_short(len(self.kerning_table))
            for item in self.kerning_table:
                item.write(buffer)


@dataclass
class TimeAttribute(Operation):
    """Extract time-related information (seconds, hours, etc.)"""
    op_code: ClassVar[OpCode] = OpCode.TimeAttribute
    id: int = 0
    time_id: int = 0
    type: int = 0
    args: List[int] = field(default_factory=list)

    def read_fields(self, buffer: WireBuffer) -> None:
        self.id = buffer.read_int()
        self.time_id = buffer.read_int()
        self.type = buffer.read_short()
        _args_count = buffer.read_short()
        if _args_count < 0:
            raise ValueError(f"Invalid negative array length for args: {_args_count}")
        if _args_count > 32:
            raise ValueError(f"Array length {_args_count} for args exceeds maximum (32)")
        if _args_count > buffer.remaining() // 4:
            raise EOFError(f"Array length {_args_count} for args exceeds remaining buffer ({buffer.remaining()})")
        self.args = [buffer.read_int() for _ in range(_args_count)]

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_int(self.id)
        buffer.write_int(self.time_id)
        buffer.write_short(self.type)
        buffer.write_short(len(self.args))
        for item in self.args:
            buffer.write_int(item)


@dataclass
class ConditionalOperations(Operation):
    """Execute a list of operations if a condition is met"""
    op_code: ClassVar[OpCode] = OpCode.ConditionalOperations
    type: int = 0
    var_a: float = 0.0
    var_b: float = 0.0

    def read_fields(self, buffer: WireBuffer) -> None:
        self.type = buffer.read_byte()
        if self.type not in (0, 1, 2, 3, 4, 5, 6):
            raise ValueError(f"Invalid enum value for type: {self.type}")
        self.var_a = buffer.read_nan_id()
        self.var_b = buffer.read_nan_id()

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        buffer.write_byte(self.type)
        buffer.write_nan_id(self.var_a)
        buffer.write_nan_id(self.var_b)


@dataclass
class BitmapTextMeasure(Operation):
    """Measure text dimensions specifically for bitmap fonts"""
    op_code: ClassVar[OpCode] = OpCode.BitmapTextMeasure
    id: int = 0
    glyph_spacing: Optional[float] = None
    text_id: int = 0
    bitmap_font_id: int = 0
    type: int = 0

    def read_fields(self, buffer: WireBuffer) -> None:
        _raw_id = buffer.read_int()
        self.id = (_raw_id & 0x7FFFFFFF)
        has_glyph_spacing = (((_raw_id & 0xFFFFFFFF) >> 31) & 0x1)
        if has_glyph_spacing != 0:
            self.glyph_spacing = buffer.read_nan_id()
        else:
            self.glyph_spacing = None
        self.text_id = buffer.read_int()
        self.bitmap_font_id = buffer.read_int()
        self.type = buffer.read_int()

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        has_glyph_spacing = (1 if self.glyph_spacing is not None else 0)
        buffer.write_int((((((self.id & 0x7FFFFFFF)) | (((has_glyph_spacing & 0x1) << 31))) & 0xFFFFFFFF) - 0x100000000 if ((((self.id & 0x7FFFFFFF)) | (((has_glyph_spacing & 0x1) << 31))) & 0xFFFFFFFF) >= 0x80000000 else ((((self.id & 0x7FFFFFFF)) | (((has_glyph_spacing & 0x1) << 31))) & 0xFFFFFFFF)))
        if has_glyph_spacing != 0:
            if self.glyph_spacing is None:
                raise ValueError("Missing required field glyphSpacing")
            buffer.write_nan_id(self.glyph_spacing)
        buffer.write_int(self.text_id)
        buffer.write_int(self.bitmap_font_id)
        buffer.write_int(self.type)


@dataclass
class ContainerEnd(Operation):
    """Closes the current container or layout component"""
    op_code: ClassVar[OpCode] = OpCode.ContainerEnd

    def read_fields(self, buffer: WireBuffer) -> None:
        pass

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)


@dataclass
class TextStyle(Operation):
    """Text style implementation"""
    op_code: ClassVar[OpCode] = OpCode.TextStyle
    id: Optional[int] = None
    font_size: Optional[float] = None
    font_axis: Optional[List[int]] = None

    def read_fields(self, buffer: WireBuffer) -> None:
        self.id = None
        self.font_size = None
        self.font_axis = None
        _sparse_count = buffer.read_short() & 0xFFFF
        if _sparse_count > buffer.remaining() // 2:
            raise EOFError(f"Sparse parameter count {_sparse_count} exceeds remaining buffer ({buffer.remaining()})")
        for _ in range(_sparse_count):
            _param_id = buffer.read_byte()
            if _param_id == 1:
                self.id = buffer.read_int()
            elif _param_id == 5:
                self.font_size = buffer.read_nan_id()
            elif _param_id == 20:
                _font_axis_count = (buffer.read_short() & 0xFFFF)
                if _font_axis_count < 0:
                    raise ValueError(f"Invalid negative array length for font_axis: {_font_axis_count}")
                if _font_axis_count > buffer.remaining() // 4:
                    raise EOFError(f"Array length {_font_axis_count} for font_axis exceeds remaining buffer ({buffer.remaining()})")
                self.font_axis = [buffer.read_int() for _ in range(_font_axis_count)]
            else:
                raise ValueError(f"Unknown sparse parameter ID: {_param_id}")

    def write(self, buffer: WireBuffer) -> None:
        buffer.write_byte(self.op_code.value)
        _sparse_count = 0
        if self.id is not None:
            _sparse_count += 1
        if self.font_size is not None:
            _sparse_count += 1
        if self.font_axis is not None:
            _sparse_count += 1
        buffer.write_short(_sparse_count)
        if self.id is not None:
            buffer.write_byte(1)
            buffer.write_int(self.id)
        if self.font_size is not None:
            buffer.write_byte(5)
            buffer.write_nan_id(self.font_size)
        if self.font_axis is not None:
            buffer.write_byte(20)
            buffer.write_short(len(self.font_axis))
            for item in self.font_axis:
                buffer.write_int(item)


OPERATION_REGISTRY: Dict[OpCode, Type[Operation]] = {
    OpCode.Header: Header,
    OpCode.DrawCircle: DrawCircle,
    OpCode.FloatExpression: FloatExpression,
    OpCode.BitmapData: BitmapData,
    OpCode.TextData: TextData,
    OpCode.BitmapFontData: BitmapFontData,
    OpCode.TimeAttribute: TimeAttribute,
    OpCode.ConditionalOperations: ConditionalOperations,
    OpCode.BitmapTextMeasure: BitmapTextMeasure,
    OpCode.ContainerEnd: ContainerEnd,
    OpCode.TextStyle: TextStyle,
}


def create_operation(op_code: OpCode) -> Optional[Operation]:
    cls = OPERATION_REGISTRY.get(op_code)
    return cls() if cls is not None else None


@dataclass
class RemoteDocument:
    """In-memory representation of a RemoteCompose document."""
    MAX_HEADER_PROPERTIES: ClassVar[int] = 1000
    MAX_OPERATION_COUNT: ClassVar[int] = 100_000
    major_version: int = 1
    minor_version: int = 1
    patch_version: int = 0
    width: int = 0
    height: int = 0
    capabilities: int = 0
    operations: List[Operation] = field(default_factory=list)

    @classmethod
    def load(cls, data: bytes, max_operations: int = MAX_OPERATION_COUNT) -> RemoteDocument:
        return cls._load_impl(data, max_operations, False, 0, 0)

    @classmethod
    def load_for_player(
        cls,
        data: bytes,
        player_api_level: int,
        player_profiles: int = 0,
        max_operations: int = MAX_OPERATION_COUNT,
    ) -> RemoteDocument:
        return cls._load_impl(data, max_operations, True, player_api_level, player_profiles)

    @classmethod
    def _load_impl(
        cls,
        data: bytes,
        max_operations: int,
        evaluate_skip: bool,
        player_api_level: int,
        player_profiles: int,
    ) -> RemoteDocument:
        buffer = WireBuffer.from_bytes(data)
        doc = cls()
        if not buffer.available():
            return doc

        header_op = buffer.read_byte()
        if header_op != OpCode.Header.value:
            raise ValueError(f"Expected Header opcode 0, got {header_op}")

        raw_major = buffer.read_int()
        doc.minor_version = buffer.read_int()
        doc.patch_version = buffer.read_int()

        if (raw_major & 0xFFFF0000) == 0x048C0000:
            doc.major_version = raw_major & 0xFFFF
            doc.width = 256
            doc.height = 256
            num_props = buffer.read_int()
            if num_props < 0 or num_props > cls.MAX_HEADER_PROPERTIES:
                raise ValueError(f"Invalid header property count: {num_props}")
            if num_props > buffer.remaining() // 4:
                raise EOFError(
                    f"Header property count {num_props} exceeds remaining buffer ({buffer.remaining()})"
                )
            for _ in range(num_props):
                tag = buffer.read_short()
                buffer.read_short()  # item_len (read and ignored)
                data_type = tag >> 10
                key = tag & 0x3F
                if data_type == 0:
                    val = buffer.read_int()
                    if key == 5:
                        doc.width = val
                    elif key == 6:
                        doc.height = val
                    elif key == 14:
                        doc.capabilities = val & 0xFFFFFFFF
                    elif key == 31 and val != 0:
                        raise ValueError("Compressed RemoteCompose documents are not supported")
                elif data_type == 1:
                    buffer.read_float()
                elif data_type == 2:
                    val = buffer.read_long()
                    if key == 14:
                        doc.capabilities = val
                    elif key == 31 and val != 0:
                        raise ValueError("Compressed RemoteCompose documents are not supported")
                elif data_type == 3:
                    buffer.read_utf8(4000)
        elif 0 <= raw_major < 0x10000:
            doc.major_version = raw_major
            doc.width = buffer.read_int()
            doc.height = buffer.read_int()
            doc.capabilities = buffer.read_long()
        else:
            raise ValueError(f"Invalid header magic number: {raw_major:#x}")

        while buffer.available():
            if len(doc.operations) >= max_operations:
                raise ValueError(f"Operation count exceeds limit {max_operations}")
            op_byte = buffer.read_byte()
            try:
                op_code = OpCode(op_byte)
            except ValueError:
                raise ValueError(f"Unknown opcode in RemoteDocument: {op_byte}") from None
            if op_code == OpCode.Header:
                raise ValueError("Unexpected Header opcode inside document body")
            op = create_operation(op_code)
            if op is None:
                raise ValueError(f"Unsupported opcode in RemoteDocument: {op_byte}")
            if evaluate_skip and hasattr(op, "read_fields_for_player"):
                op.read_fields_for_player(buffer, player_api_level, player_profiles)
            else:
                op.read_fields(buffer)
            doc.operations.append(op)

        return doc

    def save(self) -> bytes:
        buffer = WireBuffer()
        buffer.write_byte(OpCode.Header.value)
        buffer.write_int(self.major_version)
        buffer.write_int(self.minor_version)
        buffer.write_int(self.patch_version)
        buffer.write_int(self.width)
        buffer.write_int(self.height)
        buffer.write_long(self.capabilities)

        for op in self.operations:
            op.write(buffer)

        return buffer.get_bytes()
